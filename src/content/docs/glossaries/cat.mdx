{/* 1. Threat intel */}
<section id="intel" class="cat-section">
  <header class="cat-section__head">
    <span class="cat-section__ico">📡</span>
    <div>
      <h2>Advisories &amp; threat intelligence feeds</h2>
      <p>Sources of curated indicators, campaign reports, and analyst-grade intelligence.</p>
      <p class="cat-section__phases">Vocabulary for: <a href="/docs/uncover/threat-intelligence/">Uncover</a></p>
    </div>
  </header>
  <div class="cat-tools">
    <article id="abuse-ch" class="cat-tool"><h3>Abuse.ch</h3><p>Community-driven malware infrastructure tracking; free public feeds for malicious IPs, domains, and hashes. Strong on ransomware family tracking.</p></article>
    <article id="alienvault-otx" class="cat-tool"><h3>AlienVault OTX</h3><p>Collaborative threat-intel exchange aggregating IOCs from a global community, with SIEM/automated integrations.</p></article>
    <article id="anomali-threatstream" class="cat-tool"><h3>Anomali ThreatStream</h3><p>Collects, analyzes, and prioritizes threat data with enrichment and risk scoring; integrates with existing security stacks.</p></article>
    <article id="cisa-advisories" class="cat-tool"><h3>CISA Advisories</h3><p>U.S. Cybersecurity and Infrastructure Security Agency advisories on vulnerabilities and active threats, with mitigation guidance.</p></article>
    <article id="e-isac" class="cat-tool"><h3>E-ISAC</h3><p>Electricity sector ISAC; threat intel, IR support, and risk analysis tailored to grid operators.</p></article>
    <article id="fbi-flash-reports" class="cat-tool"><h3>FBI Flash Reports</h3><p>Rapid high-priority notifications about imminent cyber threats targeting public and private partners.</p></article>
    <article id="fs-isac" class="cat-tool"><h3>FS-ISAC</h3><p>Global financial-sector ISAC providing real-time alerts, incident coordination, and member collaboration.</p></article>
    <article id="greynoise" class="cat-tool"><h3>GreyNoise</h3><p>Filters internet background-scan noise so SOCs focus on targeted attacks. IP context with malicious vs. benign classification.</p></article>
    <article id="h-isac" class="cat-tool"><h3>H-ISAC</h3><p>Healthcare-sector ISAC; threat intelligence, incident response coordination, and best practices for clinical environments.</p></article>
    <article id="intel-471" class="cat-tool"><h3>Intel 471</h3><p>Commercial deep/dark-web monitoring with adversary infrastructure, TTP, and threat-actor behavior intelligence.</p></article>
    <article id="mitre-att-ck" class="cat-tool"><h3>MITRE ATT&amp;CK</h3><p>Global knowledge base of adversary tactics, techniques, and procedures. The shared vocabulary of modern SOCs.</p></article>
    <article id="ms-isac" class="cat-tool"><h3>MS-ISAC</h3><p>Multi-State ISAC for U.S. state, local, tribal, and territorial governments; alerts, IR, and risk management resources.</p></article>
    <article id="ncsc-uk" class="cat-tool"><h3>NCSC (UK)</h3><p>UK National Cyber Security Centre; guidance, threat intelligence, and IR services across public and private sectors.</p></article>
    <article id="microsoft-defender-threat-intelligence-mdti" class="cat-tool"><h3>Microsoft Defender Threat Intelligence (MDTI)</h3><p>Successor to RiskIQ PassiveTotal (acquired by Microsoft, 2021). Aggregates passive DNS, WHOIS, SSL, and infrastructure data for attribution and threat hunting.</p></article>
    <article id="phishtank" class="cat-tool"><h3>PhishTank</h3><p>Community-validated database of phishing URLs; supplies security tools and browsers with up-to-date intelligence.</p></article>
    <article id="recorded-future" class="cat-tool"><h3>Recorded Future</h3><p>Threat-intelligence platform combining machine learning with human analysis across open, dark, and technical sources.</p></article>
    <article id="microsoft-defender-easm" class="cat-tool"><h3>Microsoft Defender EASM</h3><p>External Attack Surface Management, the renamed RiskIQ Illuminate (acquired by Microsoft, 2021). Discovers and monitors internet-facing assets, third-party services, and shadow IT.</p></article>
    <article id="sans-institute" class="cat-tool"><h3>SANS Institute</h3><p>Training, certification, and research; publishes critical threat research and runs the Internet Storm Center.</p></article>
    <article id="virustotal" class="cat-tool"><h3>VirusTotal</h3><p>Free analysis of files and URLs across many AV engines; community sharing and reputation data.</p></article>
    <article id="whois" class="cat-tool"><h3>WHOIS</h3><p>Protocol and database of domain registration data; used to investigate suspicious domains and attribution.</p></article>
  </div>
</section>

    <span class="cat-section__ico">☁️</span>
    <div>
      <h2>Cloud &amp; infrastructure monitoring / security</h2>
      <p>Platforms that watch cloud workloads, identities, and posture across multi-account, multi-cloud environments.</p>

    </div>
  </header>
  <div class="cat-tools">
    <article id="aws-cloudtrail" class="cat-tool"><h3>AWS CloudTrail</h3><p>Records and logs API activity within an AWS account, the audit trail for security and forensic work.</p></article>
    <article id="aws-cloudwatch" class="cat-tool"><h3>AWS CloudWatch</h3><p>Monitoring and observability for AWS resources and applications; metrics, logs, alarms, automation hooks.</p></article>
    <article id="azure-ad-entra-id" class="cat-tool"><h3>Azure AD (Entra ID)</h3><p>Microsoft's cloud identity and access management; SSO, MFA, conditional access, and integration with SaaS apps.</p></article>
    <article id="cloudflare-cloudforce-one" class="cat-tool"><h3>Cloudflare Cloudforce One</h3><p>Cloudflare's threat-research team and intelligence service; produces actionable indicators, campaign tracking, and disruption support against active adversaries.</p></article>
    <article id="forcepoint" class="cat-tool"><h3>Forcepoint</h3><p>DLP, cloud security, and insider-threat behavior analytics across endpoints, networks, and cloud applications.</p></article>
    <article id="google-security-operations-formerly-chronicle" class="cat-tool"><h3>Google Security Operations (formerly Chronicle)</h3><p>Cloud-native security analytics platform from Google Cloud; high-speed search and threat detection at petabyte scale. Chronicle was folded into the Google Security Operations (Google SecOps) brand in 2024.</p></article>
    <article id="microsoft-sentinel" class="cat-tool"><h3>Microsoft Sentinel</h3><p>Cloud-native SIEM/SOAR (formerly Azure Sentinel); AI-driven analytics across on-prem, cloud, and hybrid with rich connector library, plus a built-in UEBA module for user and entity behavior analytics.</p></article>
    <article id="orca-security" class="cat-tool"><h3>Orca Security</h3><p>Agentless cloud security; full-stack visibility, vulnerability management, and risk prioritization across cloud workloads.</p></article>
    <article id="prisma-cloud" class="cat-tool"><h3>Prisma Cloud</h3><p>Cloud-native security platform from Palo Alto Networks; workload protection, IaC scanning, compliance, runtime defense.</p></article>
    <article id="sysdig" class="cat-tool"><h3>Sysdig</h3><p>Cloud-native security focused on container runtime, Kubernetes monitoring, and cloud workload protection.</p></article>
    <article id="wiz" class="cat-tool"><h3>WIZ</h3><p>Agentless cloud risk visibility; correlates infra/config/workload data to map attack paths and prioritize risk.</p></article>
  </div>
</section>

  <header class="cat-section__head">

    <div>
      <h2>Data collection / logging / SIEM</h2>
      <p>Where the logs go to be aggregated, normalized, and queried. The SOC's primary lens.</p>

    <article id="cribl" class="cat-tool"><h3>Cribl</h3><p>Data routing and processing for observability pipelines; filter, enrich, and reduce log data before forwarding.</p></article>
    <article id="elastic-security" class="cat-tool"><h3>Elastic Security</h3><p>Unified security on the Elastic Stack; endpoint protection, prebuilt detections, and customizable monitoring.</p></article>
    <article id="exabeam" class="cat-tool"><h3>Exabeam</h3><p>Next-gen SIEM with built-in UEBA; user-behavior timelines and automated investigation workflows.</p></article>
    <article id="fluentd" class="cat-tool"><h3>Fluentd</h3><p>Open-source data collector that unifies log collection and processing across distributed systems via plugins.</p></article>
    <article id="ibm-qradar" class="cat-tool"><h3>IBM QRadar</h3><p>Enterprise SIEM with real-time correlation, ML-assisted prioritization, and broad ecosystem integration.</p></article>
    <article id="logstash" class="cat-tool"><h3>Logstash</h3><p>Server-side pipeline that ingests, transforms, and forwards data; central to the Elastic Stack.</p></article>
    <article id="siem-concept" class="cat-tool"><h3>SIEM (concept)</h3><p>Aggregation and correlation of security event data for centralized detection, compliance, and IR.</p></article>
    <article id="sigma" class="cat-tool"><h3>Sigma</h3><p>Open, vendor-neutral detection-rule format: write a detection once in YAML, convert it to the query language of whatever SIEM is in use. The de facto standard for sharing detection logic between teams and tools.</p><a class="g-cite" href="https://github.com/SigmaHQ/sigma" target="_blank" rel="noopener noreferrer">Source: SigmaHQ</a></article>
    <article id="splunk-enterprise-security" class="cat-tool"><h3>Splunk Enterprise Security</h3><p>Premium SIEM on the Splunk platform; correlation searches, ML, prebuilt content for SOC workflows.</p></article>
    <article id="sysmon" class="cat-tool"><h3>Sysmon</h3><p>Windows service that produces rich endpoint telemetry, process creation, network connections, file events, for SIEM ingestion.</p></article>
  </div>
</section>

  <header class="cat-section__head">
    <span class="cat-section__ico">💻</span>
    <div>
      <h2>Endpoint detection &amp; response / endpoint security</h2>
      <p>The agents that watch process trees, file activity, and on-host behavior.</p>

  </header>

    <article id="bitdefender-gravityzone" class="cat-tool"><h3>Bitdefender GravityZone</h3><p>EDR with behavior analysis, anti-exploit, root-cause analysis, and policy-based remediation.</p></article>
    <article id="crowdstrike-falcon" class="cat-tool"><h3>CrowdStrike Falcon</h3><p>Cloud-native EDR with a lightweight agent; combines behavioral analytics, threat intel, and ML.</p></article>
    <article id="cybereason" class="cat-tool"><h3>Cybereason</h3><p>EDR/XDR focused on endpoint visibility, malware detection, IR, and deep process-tree forensics.</p></article>
    <article id="darktrace" class="cat-tool"><h3>Darktrace</h3><p>Self-learning AI; builds behavioral baselines, detects anomalies, autonomously responds via its RESPOND module (formerly Antigena).</p></article>

    <article id="malwarebytes-edr" class="cat-tool"><h3>Malwarebytes EDR</h3><p>Lightweight EDR with strong remediation; ransomware rollback and post-infection recovery.</p></article>
    <article id="microsoft-defender" class="cat-tool"><h3>Microsoft Defender</h3><p>Antivirus, endpoint protection, EDR (Defender for Endpoint), and XDR ties to Sentinel and Entra ID.</p></article>
    <article id="sentinelone" class="cat-tool"><h3>SentinelOne</h3><p>Unified EPP/EDR with behavioral AI, autonomous response, rollback on Windows, and threat-hunting telemetry.</p></article>
    <article id="sophos-intercept-x" class="cat-tool"><h3>Sophos Intercept X</h3><p>Next-gen AV plus EDR with deep learning and exploit prevention; strong investigation tools for SMB and enterprise.</p></article>
    <article id="symantec-broadcom" class="cat-tool"><h3>Symantec (Broadcom)</h3><p>Endpoint protection, email security, DLP, and EDR; global sensor network feeding detection and response.</p></article>
    <article id="trellix-endpoint-security" class="cat-tool"><h3>Trellix Endpoint Security</h3><p>Part of the broader Trellix XDR; detection, investigation, and response with behavioral analytics and forensic timelines.</p></article>
    <article id="trend-micro-vision-one" class="cat-tool"><h3>Trend Micro Vision One</h3><p>XDR correlating telemetry across email, endpoint, server, and cloud workloads.</p></article>
    <article id="xdr-concept" class="cat-tool"><h3>XDR (concept)</h3><p>Extended detection and response: correlates telemetry across endpoint, identity, email, network, and cloud into unified detections and response actions, extending EDR beyond the single host.</p></article>
  </div>
</section>

</section>

{/* 6. IAM */}
<section id="iam" class="cat-section">

    <span class="cat-section__ico">🔑</span>

      <h2>Identity &amp; access management / privilege management</h2>

        catRoot.style.removeProperty('--g-jump-offset');
      }
    };
    setJumpOffset();
    // This script sits before the page's style block, so on the initial
    // parse the nav is not sticky yet; re-measure once parsing finishes.
    if (document.readyState === 'loading') {
      _reg(document, 'DOMContentLoaded', setJumpOffset, { once: true });
    }
    _reg(window, 'resize', setJumpOffset);
    if ('ResizeObserver' in window) {
      const jumpRO = new ResizeObserver(setJumpOffset);
      jumpRO.observe(jumpNav);
      _cleanups.push(() => jumpRO.disconnect());
    }

  if (catRoot) {
    const toolSearch = catRoot.querySelector('[data-cat-search]');
    const toolResults = catRoot.querySelector('[data-cat-results]');
    const toolEmpty = catRoot.querySelector('[data-cat-empty]');
    const toolCards = Array.from(catRoot.querySelectorAll('.cat-tool'));
    const toolSections = Array.from(catRoot.querySelectorAll('.cat-section'));
    const filterTools = (q) => {
      const query = q.trim().toLowerCase();
      let total = 0;
      toolCards.forEach((card) => {
        const match = query === '' || card.textContent.toLowerCase().includes(query);
        card.style.display = match ? '' : 'none';
        if (match) total++;
      });
      toolSections.forEach((sec) => {
        const cards = sec.querySelectorAll('.cat-tool');
        if (cards.length === 0) return;
        const anyVisible = Array.prototype.some.call(cards, (c) => c.style.display !== 'none');
        sec.style.display = query !== '' && !anyVisible ? 'none' : '';
      });
      if (toolEmpty) toolEmpty.hidden = total > 0 || query === '';
      if (toolResults) {

      });

  // in the docs layout footer and persists in localStorage).
  const eggsEnabled = () => {
    try { return localStorage.getItem('assured-eggs') !== 'off'; } catch (e) { return true; }

  // MEOW sequence loop also checks the flag before starting a new walk.
  const animsPaused = () => catPage.classList.contains('cat-anims-paused');
  const animToggle = document.querySelector('[data-cat-anim-toggle]');
  const setAnimsPaused = (paused, persist) => {
    catPage.classList.toggle('cat-anims-paused', paused);
    if (animToggle) {
      const icon = animToggle.querySelector('.cat-anim-toggle__icon');
      const label = animToggle.querySelector('.cat-anim-toggle__text');
      if (icon) icon.textContent = paused ? '▶️' : '⏸️';
      if (label) label.textContent = paused ? 'Play animations' : 'Pause animations';
    }
    if (persist) {
      try { localStorage.setItem('assured-cat-anims', paused ? 'paused' : 'playing'); } catch (e) {}
    }
  };
  if (animToggle) {
    _reg(animToggle, 'click', () => setAnimsPaused(!animsPaused(), true));
  }
  const catReduceMq = window.matchMedia('(prefers-reduced-motion: reduce)');
  const storedAnims = (() => { try { return localStorage.getItem('assured-cat-anims'); } catch (e) { return null; } })();
  if (storedAnims === 'paused' || (storedAnims === null && catReduceMq.matches)) setAnimsPaused(true, false);
  _reg(catReduceMq, 'change', () => { if (catReduceMq.matches) setAnimsPaused(true, false); });

  bubble.style.opacity = '0';

  const playSequence = () => {
    if (animsPaused()) return;
    cat.classList.remove('cat-cat--seq');
    bubble.style.opacity = '0';
    void cat.getBoundingClientRect();

    const walks = 2 + Math.floor(Math.random() * 4);
    cat.style.setProperty('--cat-walks', walks);
    cat.classList.add('cat-cat--seq');

    const bubbleDelay = ((walks * 1.3) + 2.6) * 1000;
    setTimeout(() => {
      const isWoof = Math.random() < 0.05;
      text.textContent = isWoof ? 'WOOF' : 'MEOW';
      bubble.style.opacity = '1';
      bubble.classList.remove('cat-bubble--pop');
      void bubble.getBoundingClientRect();
      bubble.classList.add('cat-bubble--pop');
      if (isWoof) {
        const t = document.createElement('div');
        t.className = 'cat-toast';
        t.textContent = "🐶 wait, that's not a cat. you found the 5% WOOF!";
        document.querySelector('.cat-page').appendChild(t);
        setTimeout(() => t.classList.add('is-out'), 3600);
        setTimeout(() => t.remove(), 4200);
      }
      // Bubble fades out after ~2.5s so it doesn't linger until the next loop.
      // We MUST remove the cat-bubble--pop class first; its CSS animation
      // uses the forwards fill-mode, which holds the final keyframe at
      // opacity 1 and would override the inline opacity 0 below. With the
      // class removed, the cat-bubble transition (opacity 0.4s ease)
      // handles a smooth fade.
      setTimeout(() => {
        bubble.classList.remove('cat-bubble--pop');
        bubble.style.opacity = '0';
      }, 2500);
    }, bubbleDelay);
  };

  const loop = () => {
    playSequence();
    _setTimeout(loop, 18000);
  };
  if (document.readyState === 'loading') {
    _reg(document, 'DOMContentLoaded', loop);
  } else {
    loop();
  }

  // Easter egg: type "treat" to make the cat purr
  let typed = '';
  _reg(document, 'keydown', (e) => {
    if (!eggsEnabled()) return;
    if (e.key && e.key.length === 1) {
      typed = (typed + e.key.toLowerCase()).slice(-8);
      if (typed.endsWith('treat')) {
        typed = '';
        text.textContent = 'PURR';
        bubble.style.opacity = '1';
        bubble.classList.remove('cat-bubble--pop');
        void bubble.getBoundingClientRect();
        bubble.classList.add('cat-bubble--pop');
        const t = document.createElement('div');
        t.className = 'cat-toast';
        t.textContent = '🐟 you fed the cat, purrs activated';
        document.querySelector('.cat-page').appendChild(t);
        setTimeout(() => t.classList.add('is-out'), 3600);
        setTimeout(() => t.remove(), 4200);
      }
    }
  });
})();
